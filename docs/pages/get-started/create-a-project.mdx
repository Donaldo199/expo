// File: App.jsx import React, { useState, useEffect } from "react"; import { Button } from "@/components/ui/button";

const symbols = ["🍒", "🔔", "🍋", "⭐", "💎"];

const getRandomSymbol = () => symbols[Math.floor(Math.random() * symbols.length)];

export default function SlotMachine() { const [reels, setReels] = useState(["🍒", "🍒", "🍒"]); const [spinning, setSpinning] = useState(false); const [balance, setBalance] = useState(100); const [message, setMessage] = useState(""); const bet = 10;

const spin = () => { if (balance < bet || spinning) return; setSpinning(true); setMessage(""); setBalance((b) => b - bet);

const interval = setInterval(() => {
  setReels([getRandomSymbol(), getRandomSymbol(), getRandomSymbol()]);
}, 100);

setTimeout(() => {
  clearInterval(interval);
  const final = [getRandomSymbol(), getRandomSymbol(), getRandomSymbol()];
  setReels(final);
  setSpinning(false);

  if (final[0] === final[1] && final[1] === final[2]) {
    const winAmount = bet * 5;
    setBalance((b) => b + winAmount);
    setMessage(`🎉 You won ${winAmount} coins!`);
  } else {
    setMessage("😢 No match. Try again!");
  }
}, 2000);

};

return ( <div className="flex flex-col items-center justify-center min-h-screen bg-green-800 text-white"> <h1 className="text-4xl font-bold mb-6">🎰 Slot Machine</h1> <div className="flex space-x-4 text-6xl bg-black px-8 py-4 rounded-xl shadow-lg"> {reels.map((symbol, i) => ( <span key={i}>{symbol}</span> ))} </div> <p className="mt-4 text-xl">Balance: {balance} coins</p> <Button className="mt-6" onClick={spin} disabled={spinning || balance < bet}> {spinning ? "Spinning..." : Spin (-${bet})} </Button> <p className="mt-4 text-lg">{message}</p> </div> ); }
